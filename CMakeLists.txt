cmake_minimum_required(VERSION 3.24)
set(FUSE_ARCH "sm90" CACHE STRING "Architecture family: sm90 or sm103")
set_property(CACHE FUSE_ARCH PROPERTY STRINGS sm90 sm103)
if(FUSE_ARCH STREQUAL "sm90")
  set(CMAKE_CUDA_ARCHITECTURES 90a)
elseif(FUSE_ARCH STREQUAL "sm103")
  set(CMAKE_CUDA_ARCHITECTURES 103a)
else()
  message(FATAL_ERROR "Unsupported FUSE_ARCH=${FUSE_ARCH}; choose sm90 or sm103")
endif()
project(fuse_attn_cp LANGUAGES CXX CUDA)
if(FUSE_ARCH STREQUAL "sm90")
  include(cmake/sm90.cmake)
else()
  include(cmake/sm103.cmake)
endif()
